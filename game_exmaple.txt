library ieee;
use ieee.std_logic_1164.all;


entity game_example is
port(
	RST_n : in std_logic;
	CLK   : in std_logic;		-- 40 MHz
    
   PS2_CLK		: in std_logic;
	PS2_DATA	: in std_logic;
	
	VGA_H : out std_logic;
	VGA_V : out std_logic;
	VGA_R : out std_logic;
	VGA_G : out std_logic;
	VGA_B : out std_logic
);
end game_example;

architecture a of game_example is

	component vga_driver is
	port(
		rst_n	: in std_logic;
		clk	: in std_logic;
		
		animate	: out std_logic;
		xpos	: out integer;
		ypos	: out integer;
		data	: in std_logic_vector(2 downto 0);
		
		vga_h	: out std_logic;
		vga_v	: out std_logic;
		vga_r	: out std_logic;
		vga_g	: out std_logic;
		vga_b	: out std_logic
	);
	end component;

	component vga_display is
	port(
		rst_n	: in std_logic;
		clk		: in std_logic;
	
		move_left	: in std_logic;
		move_right	: in std_logic;
			move_center	: in std_logic;
			
		xpos : in integer;
		ypos : in integer;
		data : out std_logic_vector(2 downto 0);
		qidong	: in std_logic
	);
	end component;
	
	component keyboard_ctrl is
	port(
		rst_n		: in std_logic;
		clk			: in std_logic;
		animate		: in std_logic;
		ps2_clk		: in std_logic;
		ps2_data	: in std_logic;
	   move_center	: out std_logic;
		move_left	: out std_logic;
		move_right	: out std_logic;
		qidong	: out std_logic
	);
	end component;
			
	signal xpos 	: integer;
	signal ypos 	: integer;
	signal animate	: std_logic;
	signal data 	: std_logic_vector(2 downto 0);
	signal move_left	: std_logic;
	signal move_right	: std_logic;
	signal move_center	: std_logic;
	signal qidong	: std_logic;

begin

	U1 : vga_driver
	port map(
		rst_n 	=> RST_n,
		clk		=> CLK,
		animate	=> animate,
		xpos	=> xpos,
		ypos	=> ypos,
		data	=> data,		
		vga_h 	=> vga_h,
		vga_v 	=> vga_v,
		vga_r 	=> vga_r,
		vga_g 	=> vga_g,
		vga_b 	=> vga_b
	);
	
	U2 : vga_display
	port map(
		rst_n	=> RST_n,
		clk		=> CLK,
      move_center	=> move_center,
		move_left	=> move_left,
		move_right	=> move_right,
		xpos		=> xpos,
		ypos		=> ypos,
		data	=> data,
		qidong	=> qidong
	);
	
	U3 : keyboard_ctrl
	port map(
		rst_n		=> RST_n,
		clk			=> CLK,
		animate		=> animate,
		ps2_clk		=> PS2_CLK,
		ps2_data	=> PS2_DATA,
		move_left	=> move_left,
		move_center	=> move_center,
		move_right	=> move_right,
		qidong	=> qidong
	);	

end a;
