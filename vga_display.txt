library ieee;
use ieee.std_logic_1164.all;

entity vga_display is
port(
    rst_n      : in std_logic;
    clk        : in std_logic;
    xpos       : in integer;
    ypos       : in integer;
    move_left  : in std_logic;
    move_right : in std_logic;
	 move_center : in std_logic;
    data       : out std_logic_vector(2 downto 0);
    qidong     : in std_logic
);
end vga_display;

architecture a of vga_display is

    constant h_disp : integer := 800;
    constant v_disp : integer := 600;

    constant black  : std_logic_vector(2 downto 0) := "000";
    constant white  : std_logic_vector(2 downto 0) := "111";
    constant purple : std_logic_vector(2 downto 0) := "101";
    constant blue   : std_logic_vector(2 downto 0) := "001";
	 constant cyan   : std_logic_vector(2 downto 0) := "011";

    signal note_1_y : integer := 0;  -- 音符 1 的垂直位置
    signal note_2_y : integer := 0;  -- 音符 2 的垂直位置
    signal note_3_y : integer := 0;  -- 音符 3 的垂直位置

    signal note_1_x : integer := 73;   -- 音符 1 的水平位置（固定）
    signal note_2_x : integer := 340;  -- 音符 2 的水平位置（固定）
    signal note_3_x : integer := 605;  -- 音符 3 的水平位置（固定）

    signal score : integer := 0;

    -- 每个音符独立的速度计数器
    signal drop_counter_1 : integer := 0;
    signal drop_counter_2 : integer := 0;
    signal drop_counter_3 : integer := 0;

    -- 每个音符独立的速度常量
    constant drop_speed_1 : integer := 300000; -- 音符 1 掉落速度
    constant drop_speed_2 : integer := 500000; -- 音符 2 掉落速度
    constant drop_speed_3 : integer := 400000; -- 音符 3 掉落速度

begin

    -- 音符掉落逻辑
    process(clk, rst_n)
    begin
        if rst_n = '0' then
            note_1_y <= 0;
            note_2_y <= 0;
            note_3_y <= 0;
            drop_counter_1 <= 0;
            drop_counter_2 <= 0;
            drop_counter_3 <= 0;
        elsif rising_edge(clk) then
            -- 音符 1 掉落
            if drop_counter_1 < drop_speed_1 then
                drop_counter_1 <= drop_counter_1 + 1;
            else
                drop_counter_1 <= 0;  -- 重置计数器
                if note_1_y < v_disp then
                    note_1_y <= note_1_y + 1;  -- 向下移动
                else
                    note_1_y <= 0;  -- 重置位置
                end if;
            end if;

            -- 音符 2 掉落
            if drop_counter_2 < drop_speed_2 then
                drop_counter_2 <= drop_counter_2 + 1;
            else
                drop_counter_2 <= 0;  -- 重置计数器
                if note_2_y < v_disp then
                    note_2_y <= note_2_y + 1;  -- 向下移动
                else
                    note_2_y <= 0;  -- 重置位置
                end if;
            end if;

            -- 音符 3 掉落
            if drop_counter_3 < drop_speed_3 then
                drop_counter_3 <= drop_counter_3 + 1;
            else
                drop_counter_3 <= 0;  -- 重置计数器
                if note_3_y < v_disp then
                    note_3_y <= note_3_y + 1;  -- 向下移动
                else
                    note_3_y <= 0;  -- 重置位置
                end if;
            end if;
        end if;
    end process;

    -- 加分逻辑
    process(clk, rst_n)
    begin
        if rst_n = '0' then
            score <= 0;
        elsif rising_edge(clk) then
            -- 音符 1 加分判断
            if note_1_y >= 570 and note_1_y < 600 and move_left = '1' and qidong = '1' then
                score <= score + 1;
            end if;

            -- 音符 2 加分判断
            if note_2_y >= 570 and note_2_y < 600 and move_center = '1' and qidong = '1' then
                score <= score + 1;
            end if;

            -- 音符 3 加分判断
            if note_3_y >= 570 and note_3_y < 600 and move_right ='1' and qidong = '1' then
                score <= score + 1;
            end if;
        end if;
    end process;

    -- 显示逻辑
    process(xpos, ypos, note_1_y, note_2_y, note_3_y)
    begin
        -- 音符 1 显示
        if xpos >= note_1_x and xpos < note_1_x + 120 and ypos >= note_1_y and ypos < note_1_y + 30 then
            data <= blue;
              
        -- 音符 2 显示
        elsif xpos >= note_2_x and xpos < note_2_x + 120 and ypos >= note_2_y and ypos < note_2_y + 30 then
            data <= blue;

        -- 音符 3 显示
        elsif xpos >= note_3_x and xpos < note_3_x + 120 and ypos >= note_3_y and ypos < note_3_y + 30 then
            data <= blue;

        -- 判定线显示
        elsif xpos >= 0 and xpos < 800 and ypos >= 570 and ypos < 600 then
            data <= white;

        -- 分割线显示
        elsif xpos >= 261 and xpos < 271 and ypos >= 0 and ypos < 600 then
            data <= purple;
        elsif xpos >= 527 and xpos < 537 and ypos >= 0 and ypos < 600 then
            data <= purple;

        -- 其他区域显示黑色
       -- 显示分数
elsif xpos >= 740 and xpos < 800 then
    -- 清空显示区域（将分数区域的区域先清空为黑色）
    data <= black;

    -- 显示十位数字
    if score / 10 = 0 then
        -- 显示数字 0
        if (xpos >= 744 and xpos < 756 and ypos >= 10 and ypos < 14) or  -- 上横线
           (xpos >= 744 and xpos < 756 and ypos >= 66 and ypos < 70) or  -- 下横线
           (xpos >= 742 and xpos < 746 and ypos >= 14 and ypos < 66) or  -- 左竖线
           (xpos >= 756 and xpos < 760 and ypos >= 14 and ypos < 66) then -- 右竖线
            data <= cyan;
        end if;
    elsif score / 10 = 1 then
        -- 显示数字 1
        if xpos >= 750 and xpos < 754 and ypos >= 10 and ypos < 70 then  -- 竖线
            data <= cyan;
        end if;
    elsif score / 10 = 2 then
        -- 显示数字 2
        if (xpos >= 744 and xpos < 756 and ypos >= 10 and ypos < 14) or  -- 上横线
           (xpos >= 744 and xpos < 756 and ypos >= 35 and ypos < 39) or  -- 中横线
           (xpos >= 744 and xpos < 756 and ypos >= 66 and ypos < 70) or  -- 下横线
           (xpos >= 758 and xpos < 760 and ypos >= 14 and ypos < 35) or  -- 右竖线
           (xpos >= 742 and xpos < 744 and ypos >= 39 and ypos < 66) then -- 左竖线
            data <= cyan;
        end if;
    elsif score / 10 = 3 then
        -- 显示数字 3
        if (xpos >= 744 and xpos < 756 and ypos >= 10 and ypos < 14) or  -- 上横线
           (xpos >= 744 and xpos < 756 and ypos >= 35 and ypos < 39) or  -- 中横线
           (xpos >= 744 and xpos < 756 and ypos >= 66 and ypos < 70) or  -- 下横线
           (xpos >= 758 and xpos < 760 and ypos >= 14 and ypos < 35) or  -- 右竖线
           (xpos >= 758 and xpos < 760 and ypos >= 39 and ypos < 66) then -- 右竖线
            data <= cyan;
        end if;
    elsif score / 10 = 4 then
        -- 显示数字 4
        if (xpos >= 742 and xpos < 744 and ypos >= 10 and ypos < 39) or  -- 左竖线
           (xpos >= 744 and xpos < 756 and ypos >= 35 and ypos < 39) or  -- 中横线
           (xpos >= 758 and xpos < 760 and ypos >= 10 and ypos < 70) then -- 右竖线
            data <= cyan;
        end if;
    elsif score / 10 = 5 then
        -- 显示数字 5
        if (xpos >= 744 and xpos < 756 and ypos >= 10 and ypos < 14) or  -- 上横线
           (xpos >= 744 and xpos < 756 and ypos >= 35 and ypos < 39) or  -- 中横线
           (xpos >= 744 and xpos < 756 and ypos >= 66 and ypos < 70) or  -- 下横线
           (xpos >= 742 and xpos < 746 and ypos >= 14 and ypos < 35) or  -- 左竖线
           (xpos >= 756 and xpos < 760 and ypos >= 39 and ypos < 66) then -- 右竖线
            data <= cyan;
        end if;
    elsif score / 10 = 6 then
        -- 显示数字 6
        if (xpos >= 744 and xpos < 756 and ypos >= 10 and ypos < 14) or  -- 上横线
           (xpos >= 744 and xpos < 756 and ypos >= 35 and ypos < 39) or  -- 中横线
           (xpos >= 744 and xpos < 756 and ypos >= 66 and ypos < 70) or  -- 下横线
           (xpos >= 742 and xpos < 746 and ypos >= 14 and ypos < 66) or  -- 左竖线
           (xpos >= 756 and xpos < 760 and ypos >= 39 and ypos < 66) then -- 右竖线
            data <= cyan;
        end if;
    elsif score / 10 = 7 then
        -- 显示数字 7
        if (xpos >= 744 and xpos < 756 and ypos >= 10 and ypos < 14) or  -- 上横线
           (xpos >= 758 and xpos < 760 and ypos >= 14 and ypos < 70) then -- 右竖线
            data <= cyan;
        end if;
    elsif score / 10 = 8 then
        -- 显示数字 8
        if (xpos >= 744 and xpos < 756 and ypos >= 10 and ypos < 14) or  -- 上横线
           (xpos >= 744 and xpos < 756 and ypos >= 35 and ypos < 39) or  -- 中横线
           (xpos >= 744 and xpos < 756 and ypos >= 66 and ypos < 70) or  -- 下横线
           (xpos >= 742 and xpos < 746 and ypos >= 14 and ypos < 66) or  -- 左竖线
           (xpos >= 756 and xpos < 760 and ypos >= 14 and ypos < 66) then -- 右竖线
            data <= cyan;
        end if;
    elsif score / 10 = 9 then
        -- 显示数字 9
        if (xpos >= 744 and xpos < 756 and ypos >= 10 and ypos < 14) or  -- 上横线
           (xpos >= 744 and xpos < 756 and ypos >= 35 and ypos < 39) or  -- 中横线
           (xpos >= 744 and xpos < 756 and ypos >= 66 and ypos < 70) or  -- 下横线
           (xpos >= 758 and xpos < 760 and ypos >= 14 and ypos < 35) or  -- 右竖线
           (xpos >= 742 and xpos < 746 and ypos >= 14 and ypos < 35) or  -- 左竖线
           (xpos >= 758 and xpos < 760 and ypos >= 39 and ypos < 66) then -- 右竖线
            data <= cyan;
        end if;
    end if;

    -- 显示个位数字
    if score mod 10 = 0 then
        -- 显示数字 0
        if (xpos >= 784 and xpos < 796 and ypos >= 10 and ypos < 14) or  -- 上横线
           (xpos >= 784 and xpos < 796 and ypos >= 66 and ypos < 70) or  -- 下横线
           (xpos >= 782 and xpos < 786 and ypos >= 14 and ypos < 66) or  -- 左竖线
           (xpos >= 796 and xpos < 800 and ypos >= 14 and ypos < 66) then -- 右竖线
            data <= cyan;
        end if;
    elsif score mod 10 = 1 then
        -- 显示数字 1
        if xpos >= 790 and xpos < 794 and ypos >= 10 and ypos < 70 then  -- 竖线
            data <= cyan;
        end if;
    elsif score mod 10 = 2 then
        -- 显示数字 2
        if (xpos >= 784 and xpos < 796 and ypos >= 10 and ypos < 14) or  -- 上横线
           (xpos >= 784 and xpos < 796 and ypos >= 35 and ypos < 39) or  -- 中横线
           (xpos >= 784 and xpos < 796 and ypos >= 66 and ypos < 70) or  -- 下横线
           (xpos >= 798 and xpos < 800 and ypos >= 14 and ypos < 35) or  -- 右竖线
           (xpos >= 782 and xpos < 784 and ypos >= 39 and ypos < 66) then -- 左竖线
            data <= cyan;
        end if;
    elsif score mod 10 = 3 then
    -- 显示数字 3
    if (xpos >= 784 and xpos < 796 and ypos >= 10 and ypos < 14) or  -- 上横线
       (xpos >= 784 and xpos < 796 and ypos >= 35 and ypos < 39) or  -- 中横线
       (xpos >= 784 and xpos < 796 and ypos >= 66 and ypos < 70) or  -- 下横线
       (xpos >= 798 and xpos < 800 and ypos >= 14 and ypos < 35) or  -- 右竖线
       (xpos >= 798 and xpos < 800 and ypos >= 39 and ypos < 66) then -- 右竖线
        data <= cyan;
    end if;

    elsif score mod 10 = 4 then
    -- 显示数字 4
    if (xpos >= 782 and xpos < 784 and ypos >= 10 and ypos < 39) or  -- 左竖线
       (xpos >= 784 and xpos < 796 and ypos >= 35 and ypos < 39) or  -- 中横线
       (xpos >= 798 and xpos < 800 and ypos >= 10 and ypos < 70) then -- 右竖线
        data <= cyan;
    end if;

    elsif score mod 10 = 5 then
    -- 显示数字 5
    if (xpos >= 784 and xpos < 796 and ypos >= 10 and ypos < 14) or  -- 上横线
       (xpos >= 784 and xpos < 796 and ypos >= 35 and ypos < 39) or  -- 中横线
       (xpos >= 784 and xpos < 796 and ypos >= 66 and ypos < 70) or  -- 下横线
       (xpos >= 782 and xpos < 786 and ypos >= 14 and ypos < 35) or  -- 左竖线
       (xpos >= 798 and xpos < 800 and ypos >= 39 and ypos < 66) then -- 右竖线
        data <= cyan;
    end if;

    elsif score mod 10 = 6 then
    -- 显示数字 6
    if (xpos >= 784 and xpos < 796 and ypos >= 10 and ypos < 14) or  -- 上横线
       (xpos >= 784 and xpos < 796 and ypos >= 35 and ypos < 39) or  -- 中横线
       (xpos >= 784 and xpos < 796 and ypos >= 66 and ypos < 70) or  -- 下横线
       (xpos >= 782 and xpos < 786 and ypos >= 14 and ypos < 66) or  -- 左竖线
       (xpos >= 798 and xpos < 800 and ypos >= 39 and ypos < 66) then -- 右竖线
        data <= cyan;
    end if;

    elsif score mod 10 = 7 then
    -- 显示数字 7
    if (xpos >= 784 and xpos < 796 and ypos >= 10 and ypos < 14) or  -- 上横线
       (xpos >= 798 and xpos < 800 and ypos >= 14 and ypos < 70) then -- 右竖线
        data <= cyan;
    end if;

    elsif score mod 10 = 8 then
    -- 显示数字 8
    if (xpos >= 784 and xpos < 796 and ypos >= 10 and ypos < 14) or  -- 上横线
       (xpos >= 784 and xpos < 796 and ypos >= 35 and ypos < 39) or  -- 中横线
       (xpos >= 784 and xpos < 796 and ypos >= 66 and ypos < 70) or  -- 下横线
       (xpos >= 782 and xpos < 786 and ypos >= 14 and ypos < 66) or  -- 左竖线
       (xpos >= 798 and xpos < 800 and ypos >= 14 and ypos < 66) then -- 右竖线
        data <= cyan;
    end if;

    elsif score mod 10 = 9 then
    -- 显示数字 9
    if (xpos >= 784 and xpos < 796 and ypos >= 10 and ypos < 14) or  -- 上横线
       (xpos >= 784 and xpos < 796 and ypos >= 35 and ypos < 39) or  -- 中横线
       (xpos >= 784 and xpos < 796 and ypos >= 66 and ypos < 70) or  -- 下横线
       (xpos >= 798 and xpos < 800 and ypos >= 14 and ypos < 35) or  -- 右竖线
       (xpos >= 782 and xpos < 786 and ypos >= 14 and ypos < 35) or  -- 左竖线
       (xpos >= 798 and xpos < 800 and ypos >= 39 and ypos < 66) then -- 右竖线
        data <= cyan;
    end if;
end if;
else
    data <= black;
end if;
    end process;

end a;
